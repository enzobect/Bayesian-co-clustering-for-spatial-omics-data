% Style pour les Définitions Techniques 
\newtcolorbox{definition}[1]{%
    colback=monBlue!5!white,    % Fond bleu très clair
    colframe=monBlue!75!black,  % Bordure bleu foncé
    fonttitle=\bfseries,
    title=Définition : #1,    % Titre automatique
    arc=0mm,                 % Coins carrés (plus sérieux)
    enhanced,
    drop shadow % Petit effet d'ombre
    after skip=15pt
}

% --- Style pour les Institutions ---
\newtcolorbox{institution}[1]{%
    colback=gray!10!white,   % Fond gris clair
    colframe=gray!80!black,  % Bordure grise
    fonttitle=\bfseries,
    title=Institution : #1,   % Titre automatique
    arc=3mm,                 % Coins arrondis
    boxrule=1pt,
    left=10pt      
}

% --- Style pour les Textes de Loi et Articles de Droit ---
\newtcolorbox{loi}[1]{%
    enhanced,
    colback=white,           % Fond blanc pour rester sobre
    colframe=gray!60!black,  % Bordure grise foncée
    arc=0mm,                 % Coins bien droits
    boxrule=0pt,             % Pas de bordure générale
    leftrule=4pt,            % Grosse barre à gauche
    fonttitle=\bfseries\color{gray!60!black},
    title=Réf : #1,           % Titre discret (ex: Art. 2 Convention d'Oviedo)
    detach title,            % On sépare le titre du corps
    before upper={\textbf{#1} \medskip \par}, % Affiche la réf en gras au début
    fontupper=\itshape,      % Texte de loi en italique
    after skip=15pt          % Espace après le cadre
}
